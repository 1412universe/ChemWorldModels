# 已踩坑：默认 SSH:22 不通；改 HTTPS 会卡住；唯一稳的是 SSH over 443。

cd /Users/fenglongli/Downloads/ChemWorldModel

# 强制 remote 为 SSH（若误改成 https:// 会 push 卡住，勿用 HTTPS）
git remote set-url origin git@github.com:1412universe/ChemWorldModels.git

git add -A
git status

# 有暂存改动才 commit；没有则跳过（避免「nothing to commit」中断后面的 push）
git diff --cached --quiet || git commit -m "$(cat <<'EOF'
更新说明写这里
EOF
)"

# 推送：必须走 ssh.github.com:443（不要用默认 22，也不要 git push 裸 HTTPS）
GIT_SSH_COMMAND='ssh -o Hostname=ssh.github.com -p 443 -o User=git' git push origin main

# 核对：应显示与 origin/main 对齐，不应再有 ahead
git status -sb
git log -1 --oneline
